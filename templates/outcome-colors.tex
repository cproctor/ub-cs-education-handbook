% Color definitions for learning-outcome badges (CK/PK/PCK/L), used by
% filters/outcome-colors.lua. Keep in sync with the group->class mapping in
% tasks/table_generation.py and the badge colors in handbook.css.
\definecolor{OutcomeCK}{HTML}{0EA5E9}
\definecolor{OutcomePK}{HTML}{059669}
\definecolor{OutcomePCK}{HTML}{A855F7}
\definecolor{OutcomeL}{HTML}{EA580C}

% Pill-style learning-outcome badge: a colored background with black text,
% matching the .outcome badge style in handbook.css. The fill is drawn at
% 85% opacity (over the page's white background) for a softer look, matching
% the rgba() backgrounds in handbook.css.
%
% Deliberately NOT built with tikz (as earlier versions were): tikz.sty is
% only "partially compatible" with the LaTeX tagging system used for PDF/UA-1
% accessibility (per \DocumentMetadata{check-tagging-status} in
% eisvogel.latex), and in testing, text placed via a tikz \node did not reach
% the PDF's structure tree at all -- it was visible on the page but invisible
% to screen readers. \colorbox/\textcolor are ordinary text-flow commands
% and inherit the tagging of their enclosing paragraph for free. The
% trade-off is square corners instead of the previous 2pt-rounded pill.
\newcommand{\OutcomeBadge}[2]{%
  \colorbox{#1!85}{\textcolor{black}{\bfseries\scriptsize#2}}%
}

% Colored dot marking an alignment-table cell, matching the .dot marker
% style in handbook.css. Also not built with tikz, for the same structure-
% tree reason as \OutcomeBadge above. Since the dot itself carries no visible
% text -- its presence in a cell is the information (a course addresses an
% outcome) -- it's given an explicit `actualtext` so a screen reader
% announces something for the cell instead of reading it as empty, which
% would be indistinguishable from a cell where the outcome isn't addressed.
\newcommand{\OutcomeDot}[1]{%
  \tagmcbegin{tag=Span,actualtext=Yes}%
  \textcolor{#1}{\raisebox{0.05em}{\scalebox{1.3}{\textbullet}}}%
  \tagmcend
}

% Works around a "No counter 'none' defined" error from recent versions of
% the `caption` package, which no longer tolerate pandoc's uncaptioned-table
% workaround (`\def\LTcaptype{none}`) for longtables without a caption.
\newcounter{none}
